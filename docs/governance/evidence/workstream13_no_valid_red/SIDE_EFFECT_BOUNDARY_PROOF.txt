def imports(path):
    tree = ast.parse(open(path,encoding='utf-8').read(), path); out=[]
    for n in tree.body:
        if isinstance(n, ast.Import): out += [a.name for a in n.names]
        elif isinstance(n, ast.ImportFrom): out.append(f"{n.module} -> {', '.join(a.name for a in n.names)}")
    return out
def modlevel_names(path):
    tree = ast.parse(open(path,encoding='utf-8').read(), path); names=[]
    for n in tree.body:
        if isinstance(n, ast.Assign): names += [t.id for t in n.targets if isinstance(t, ast.Name)]
        elif isinstance(n, ast.AnnAssign) and isinstance(n.target, ast.Name): names.append(n.target.id)
    return names
def calls_for(modname):
    if modname.endswith('answer_coauthoring_prompts') or modname.endswith('clarification_labels'): return [((g,),{}) for g in gaps]
    if modname.endswith('scaffolding_guidance'): return [((lr,'MECHANISM_COMPLETENESS'),{}) for lr in warn_lr]
    if modname.endswith('result_feedback'): return [((lr,),{}) for lr in warn_lr]
    if modname.endswith('uncertainty_guidance'): return [((t,),{}) for t in uq_texts]
    return []
MUT = (list, dict, set, bytearray)
for path,(modname,fnname) in FILES.items():
    m = importlib.import_module(modname); names = modlevel_names(path)
    print(f"=== {path} ===")

    after = {nm: getattr(m, nm) for nm in names}
    mutated = [nm for nm in names if before[nm] != after[nm]]
    print(f"  ANY_STRUCTURALLY_MUTABLE={any_mut}")
    print(f"  MUTATED_DURING_REPRESENTATIVE_CALLS={mutated if mutated else 'NONE'}")
